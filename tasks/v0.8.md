# Task: v0.8 — CPF Tab, Transaction History & Projection Corrections

Read `HANDOVER.md` first to understand the current state of the codebase.

## What this task builds

Three improvements to the CPF/retirement feature:

1. **CPF tab** — dedicated `CPF` tab in the Expenses page (between Transactions and Projections). Moves the CPF balance panel inside it and adds a CPF transaction history list.
2. **CPF transaction storage** — parse and persist every individual transaction from the CPF annual statement into a new `CpfTransactions` Google Sheet tab.
3. **Projection corrections** — deduct HDB loan payments from OA and MediSave premiums from MA in the year-by-year retirement projection engine.

---

## Project context

- **Framework:** Next.js 16.2.3 App Router (TypeScript, Tailwind v4)
- **Data store:** Google Sheets (no database)
- **Path alias:** `@/` maps to `src/`
- **Package manager:** npm

### Critical Next.js 16 rules

- `params` is a `Promise` in dynamic routes — always `await` it
- Every API route file must export `export const dynamic = 'force-dynamic'`
- Route Handlers use Web APIs: `Request`, `Response.json(...)` — never `NextApiRequest`/`NextApiResponse`
- No `getServerSideProps` or `getStaticProps` — App Router only
- Pages stay as Server Components; interactive UI lives in `'use client'` components at the leaves

### Critical project conventions

1. **No module-level Sheets singleton.** Always instantiate inside a local `getSheetsClient()` function.
2. **Private key transform.** Always call `.replace(/\\n/g, '\n')` on `GOOGLE_PRIVATE_KEY` inside `getSheetsClient()`.
3. **All types** imported from `@/types` — never defined inline in other files.
4. **All imports** use `@/` alias, never relative `../../`.
5. **API error shape:** `{ error: string }` with appropriate HTTP status on failure.
6. **No comments** unless the WHY is genuinely non-obvious.
7. **Server Actions** live in `src/app/lib/actions.ts` which starts with `'use server'`.
8. **Styling:** Tailwind v4. SGD format: `S$1,234.56`. Gain colour `#16a34a` (`text-green-600`), loss colour `#dc2626` (`text-red-600`).

import { google } from 'googleapis';

function getSheetsClient() {
  const {
    GOOGLE_SHEETS_SPREADSHEET_ID,
    GOOGLE_SERVICE_ACCOUNT_EMAIL,
    GOOGLE_PRIVATE_KEY,
  } = process.env;

  if (!GOOGLE_SHEETS_SPREADSHEET_ID || !GOOGLE_SERVICE_ACCOUNT_EMAIL || !GOOGLE_PRIVATE_KEY) {
    throw new Error('Missing Google Sheets credentials in environment variables.');

  const auth = new google.auth.GoogleAuth({
    credentials: {

    },
    scopes: ['https://www.googleapis.com/auth/spreadsheets'],
  });

  return {
    sheets: google.sheets({ version: 'v4', auth }),

  };
}

```ts
export const dynamic = 'force-dynamic';

import { google } from 'googleapis';

const NEW_TAB = { title: 'CpfTransactions', headers: ['date', 'code', 'ref', 'oa_change', 'sa_change', 'ma_change'] };

export async function POST(_request: Request) {
  try {
    const {
      GOOGLE_SHEETS_SPREADSHEET_ID: spreadsheetId,
      GOOGLE_SERVICE_ACCOUNT_EMAIL: serviceAccountEmail,
      GOOGLE_PRIVATE_KEY: privateKey,
    } = process.env;

    if (!spreadsheetId || !serviceAccountEmail || !privateKey) {
      return Response.json({ error: 'Missing Google Sheets credentials.' }, { status: 400 });
    }

    const auth = new google.auth.GoogleAuth({
      credentials: {
        client_email: serviceAccountEmail,
        private_key: privateKey.replace(/\\n/g, '\n'),
      },
      scopes: ['https://www.googleapis.com/auth/spreadsheets'],
    });

    const sheets = google.sheets({ version: 'v4', auth });
    const meta = await sheets.spreadsheets.get({ spreadsheetId });
    const existing = meta.data.sheets?.map(s => s.properties?.title ?? '') ?? [];

    if (existing.includes(NEW_TAB.title)) {
      return Response.json({ created: [], skipped: [NEW_TAB.title] });
    }

    await sheets.spreadsheets.batchUpdate({
      spreadsheetId,
      requestBody: { requests: [{ addSheet: { properties: { title: NEW_TAB.title } } }] },
    });

    await sheets.spreadsheets.values.update({
      spreadsheetId,
      range: `${NEW_TAB.title}!A1`,
      valueInputOption: 'RAW',
      requestBody: { values: [NEW_TAB.headers] },
    });

    return Response.json({ created: [NEW_TAB.title], skipped: [] });
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    return Response.json({ error: message }, { status: 500 });
  }
}
```

function parseSignedAmount(raw: string): number {
  const negative = raw.startsWith('-');
  return (negative ? -1 : 1) * parseFloat(raw.replace(/[-$,]/g, ''));
}

export function parseCpfTransactions(text: string): CpfTransaction[] {
  const yearMatch = text.match(/Yearly Statement of Account for (\d{4})/);
  if (!yearMatch) throw new Error('Could not find statement year');
  const year = yearMatch[1];

  const MON = 'JAN|FEB|MAR|APR|MAY|JUN|JUL|AUG|SEP|OCT|NOV|DEC';
  const txRegex = new RegExp(
    `(\\d{2})\\s*(${MON})([A-Z]{2,10})((?:[A-Z]{3} \\d{4}[A-Z])?)` +
    `(-?\\$[\\d,]+\\.\\d{2})(-?\\$[\\d,]+\\.\\d{2})(-?\\$[\\d,]+\\.\\d{2})`,
    'g',
  );

  const results: CpfTransaction[] = [];
  let m: RegExpExecArray | null;
  while ((m = txRegex.exec(text)) !== null) {
    const [, day, mon, code, ref, rawOa, rawSa, rawMa] = m;
    if (code === 'BAL') continue;
    results.push({
      date: `${year}-${MONTHS[mon]}-${day}`,
      code,
      ref: ref.trim(),
      oaChange: parseSignedAmount(rawOa),
      saChange: parseSignedAmount(rawSa),
      maChange: parseSignedAmount(rawMa),
    });
  }
  return results;
}
```

export async function getCpfTransactionKeys(): Promise<Set<string>> {
  const { sheets, spreadsheetId } = getSheetsClient();
  const res = await sheets.spreadsheets.values.get({
    spreadsheetId,
    range: `${SHEET_NAMES.CPF_TRANSACTIONS}!A:C`,
  });
  const rows = res.data.values ?? [];
  const keys = new Set<string>();
  for (let i = 1; i < rows.length; i++) {
    const r = rows[i];
    if (r[0] && r[1] != null && r[2] != null) keys.add(`${r[0]}|${r[1]}|${r[2]}`);
  }
  return keys;
}

export async function appendCpfTransactions(rows: CpfTransaction[]): Promise<void> {
  if (rows.length === 0) return;
  const { sheets, spreadsheetId } = getSheetsClient();
  await sheets.spreadsheets.values.append({
    spreadsheetId,
    range: `${SHEET_NAMES.CPF_TRANSACTIONS}!A:F`,
    valueInputOption: 'RAW',
    requestBody: {
      values: rows.map(r => [
        r.date,
        r.code,
        r.ref,
        r.oaChange.toString(),
        r.saChange.toString(),
        r.maChange.toString(),
      ]),
    },
  });
}

export async function getCpfTransactions(): Promise<CpfTransaction[]> {
  const { sheets, spreadsheetId } = getSheetsClient();
  const res = await sheets.spreadsheets.values.get({
    spreadsheetId,
    range: `${SHEET_NAMES.CPF_TRANSACTIONS}!A:F`,
  });
  const rows = res.data.values ?? [];
  if (rows.length <= 1) return [];
  return rows.slice(1)
    .filter(r => r[0] && r[1] != null && r[2] != null)
    .map(r => ({
      date: r[0],
      code: r[1],
      ref: r[2] ?? '',
      oaChange: parseFloat(r[3] ?? '0'),
      saChange: parseFloat(r[4] ?? '0'),
      maChange: parseFloat(r[5] ?? '0'),
    }))
    .sort((a, b) => a.date.localeCompare(b.date));
}
```

'use client';

import { CpfTransaction } from '@/types';

function fmtSGD(n: number): string {
  return 'S$' + Math.abs(n).toLocaleString('en-SG', { minimumFractionDigits: 2, maximumFractionDigits: 2 });

function fmtChange(n: number) {
  if (n === 0) return <span className="text-slate-300">—</span>;
  const cls = n > 0 ? 'text-green-600' : 'text-red-600';
  return <span className={cls}>{n > 0 ? '+' : ''}{fmtSGD(n)}</span>;
}

function toMonthKey(date: string) { return date.slice(0, 7); }

function formatMonth(key: string) {
  const [y, m] = key.split('-');
  return new Date(parseInt(y), parseInt(m) - 1, 1).toLocaleDateString('en-SG', { month: 'short', year: 'numeric' });

  transactions: CpfTransaction[];
